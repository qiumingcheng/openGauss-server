-- 2表JOIN索引消除ORDER BY 测试用例
-- 执行前需要创建测试表和索引

-- =============================================
-- 前提条件：创建测试表和索引
-- =============================================
-- 假设已存在 t1, t2 测试表，表结构如下：
-- t1(id int, a int, b int, c text)
-- t2(id int, a int, b int, c text)
-- idx_t1_b: b 列上的索引
-- idx_t1_ab: (a, b) 多列索引
-- idx_t1_id: id 列上的主键索引

-- =============================================
-- 场景1: ORDER BY 外表列，有可用单列索引
-- 期望: 外表 t1 使用 idx_t1_b 索引消除排序
-- =============================================
SELECT * FROM t1 JOIN t2 ON t1.id = t2.id ORDER BY t1.b;

-- =============================================
-- 场景2: ORDER BY 外表主键列
-- 期望: 外表 t1 使用主键索引 idx_t1_id 消除排序
-- =============================================
SELECT * FROM t1 JOIN t2 ON t1.id = t2.id ORDER BY t1.id;

-- =============================================
-- 场景3: ORDER BY 外表多列，使用多列索引
-- 期望: 外表 t1 使用 idx_t1_ab 多列索引消除排序
-- =============================================
SELECT * FROM t1 JOIN t2 ON t1.id = t2.id ORDER BY t1.a, t1.b;

-- =============================================
-- 场景4: ORDER BY 外表列，但索引方向不匹配（降序）
-- 期望: 可能需要 PhysicalSort 或使用索引但需要额外处理
-- =============================================
SELECT * FROM t1 JOIN t2 ON t1.id = t2.id ORDER BY t1.b DESC;

-- =============================================
-- 场景5: ORDER BY 内表列，无法消除
-- 期望: 需要 PhysicalSort 显式排序
-- =============================================
SELECT * FROM t1 JOIN t2 ON t1.id = t2.id ORDER BY t2.b;

-- =============================================
-- 场景6: RIGHT JOIN 外表列，无法消除（RIGHT JOIN破坏排序）
-- 期望: 需要 PhysicalSort 显式排序
-- =============================================
SELECT * FROM t1 RIGHT JOIN t2 ON t1.id = t2.id ORDER BY t1.b;

-- =============================================
-- 场景7: FULL JOIN 外表列，无法消除（FULL JOIN破坏排序）
-- 期望: 需要 PhysicalSort 显式排序
-- =============================================
SELECT * FROM t1 FULL JOIN t2 ON t1.id = t2.id ORDER BY t1.b;

-- =============================================
-- 场景8: LEFT JOIN 外表列，非FULL/RIGHT类型
-- 期望: 外表 t1 使用索引消除排序
-- =============================================
SELECT * FROM t1 LEFT JOIN t2 ON t1.id = t2.id ORDER BY t1.b;

-- =============================================
-- 场景9: ORDER BY 外表列 + WHERE 条件
-- 期望: 外表索引扫描 + join
-- =============================================
SELECT * FROM t1 JOIN t2 ON t1.id = t2.id WHERE t1.a > 100 ORDER BY t1.b;

-- =============================================
-- 场景10: JOIN后ORDER BY两表的混合列
-- 期望: 无法完全消除排序，需要 PhysicalSort
-- =============================================
SELECT * FROM t1 JOIN t2 ON t1.id = t2.id ORDER BY t1.b, t2.a;

-- =============================================
-- 场景11: NestLoop JOIN + ORDER BY 外表列
-- 期望: 外表 t1 使用索引消除排序，内表 t2 作为参数化扫描
-- =============================================
SELECT * FROM t1 INNER JOIN t2 ON t1.id = t2.id ORDER BY t1.b;

-- =============================================
-- 场景12: 3表JOIN，ORDER BY 第一个表的列
-- 期望: t1 使用索引消除排序
-- =============================================
SELECT * FROM t1 JOIN t2 ON t1.id = t2.id JOIN t3 ON t2.id = t3.id ORDER BY t1.b;

-- =============================================
-- 验证执行计划的方法（使用 EXPLAIN）
-- =============================================
-- 在集成测试中，可以通过 EXPLAIN 查看执行计划
-- 预期看到的关键词:
-- - "Index Scan" 或 "Index Only Scan" 表示使用了索引
-- - "Sort" 表示需要显式排序
-- - "Passthrough" 或 "Pass" 表示通过 PassThrough 消除排序

EXPLAIN SELECT * FROM t1 JOIN t2 ON t1.id = t2.id ORDER BY t1.b;
EXPLAIN SELECT * FROM t1 JOIN t2 ON t1.id = t2.id ORDER BY t2.b;
